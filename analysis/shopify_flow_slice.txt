 130     public override async Task<StepOutcome> ScrapeVariantAvailabilityAsync(FlowContext c, CancellationToken ct)
 131     {
 132         if (!c.Session.Vars.TryGetValue("raw", out var raw) || string.IsNullOrWhiteSpace(raw))
 133             return StepOutcome.Fail("shopify: no product payload");
 134 
 135         var root = JObject.Parse(raw);
 136         var variants = root["variants"] as JArray;
 137         var anyAvailable = variants is not null &&
 138                            variants.Any(v => v["available"]?.ToObject<bool?>() == true);
 139 
 140         c.Session.Vars["available"] = anyAvailable ? "true" : "false";
 141         return StepOutcome.Success();
 142     }

 146     {
 147         if (string.IsNullOrEmpty(c.Session.VariantId)) return StepOutcome.Fail("shopify: no variant to add");
 148 
 149         var payload = new JObject
 150         {
 151             ["id"] = long.TryParse(c.Session.VariantId, out var id) ? id : c.Session.VariantId,
 152             ["quantity"] = c.Session.Quantity,
 153         };
 154 
 155         var res = await PostJsonAsync(c, Origin(c.Url) + "/cart/add.js",
 156             payload.ToString(Newtonsoft.Json.Formatting.None), "shopify-cart-add", ct)
 157             .ConfigureAwait(false);
 158 
 159         if (res is null) return StepOutcome.Fail("shopify add to cart failed", retryable: true);
 160         if (res.StatusCode is 422 or 404) return StepOutcome.Fail("shopify: variant sold out");
 161         if (!res.IsSuccess) return StepOutcome.Fail($"shopify cart HTTP {res.StatusCode}", retryable: true);
 162 
 163         c.Session.Vars["cart"] = res.Body;
 164         return StepOutcome.Success();
 165     }
